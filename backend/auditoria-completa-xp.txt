      function finishQuiz() {
        setScreen('result');

        const total = perguntas.length;
        const porcentagem = total > 0
          ? Math.round((acertos / total) * 100)
          : 0;

        scoreValue.textContent = `${acertos}/${total}`;

const quizModel = require('../models/perguntas');
const userModel = require('../models/user');

const iniciarQuiz = async (req, res) => {
  try {
    const { materiaId } = req.params;
    const perguntas = await quizModel.buscarPerguntasPorMateria(materiaId);

    if (!perguntas || perguntas.length === 0) {
      return res.status(404).json({ erro: 'Nenhuma pergunta encontrada.' });
    }

    res.status(200).json({ materia: materiaId, perguntas });
  } catch (error) {
    console.error(error);
    res.status(500).json({ erro: 'Erro ao iniciar o quiz.' });
  }
};

const responderPergunta = async (req, res) => {
  try {
    const { perguntaId, resposta } = req.body;

    if (!perguntaId || resposta === undefined) {
      return res.status(400).json({ erro: 'Pergunta e resposta sÃ£o obrigatÃ³rias.' });
    }

    const resultado = await quizModel.verificarResposta(perguntaId, resposta);
    res.status(200).json(resultado);
  } catch (error) {
    console.error(error);
    res.status(500).json({ erro: 'Erro ao verificar resposta.' });
  }
};

const finalizarQuiz = async (req, res) => {
  try {
    const {
      usuarioId,
      materiaId,
      respostas,
      totalQuizzes,
      materiasExploradas,
      ultimaPontuacao
    } = req.body;

    if (!usuarioId || !materiaId || !respostas) {
      return res.status(400).json({ erro: 'Dados do resultado incompletos.' });
    }

    const resultado = await quizModel.calcularResultado(usuarioId, materiaId, respostas);
    const usuarioComXp = await userModel.atualizarXpUsuario(usuarioId, resultado.xpGanho || 0, {
      totalQuizzes,
      materiasExploradas,
      ultimaPontuacao
    });

    res.status(200).json({
      ...resultado,
      usuario: usuarioComXp,
      xpGanho: resultado.xpGanho || 0
    });
  } catch (error) {
    console.error(error);
    res.status(500).json({ erro: 'Erro ao finalizar o quiz.' });
  }
};

module.exports = {
  iniciarQuiz,
  responderPergunta,
  finalizarQuiz
};

const isSafeInternalRedirect = (value) => {
  if (typeof value !== 'string') {
    return false;
  }

  const target = value.trim();

  if (!target || target.startsWith('//') || /^([a-z]+:)?\/\//i.test(target) || /^javascript:/i.test(target) || /^data:/i.test(target)) {
    return false;
  }

  return target.startsWith('/');
};

const isApiRequest = (req) => {
  const acceptHeader = String(req.headers.accept || '');
  return req.path.startsWith('/api') || acceptHeader.includes('application/json') || req.headers['x-requested-with'] === 'XMLHttpRequest';
};

const isHtmlRequest = (req) => !isApiRequest(req) && Boolean(req.accepts('html'));

const normalizeRedirectTarget = (req) => {
  const target = req.originalUrl || req.url || '/home';

  if (!target || target === '/login' || target.startsWith('/login?')) {
    return '/home';
  }

  return isSafeInternalRedirect(target) ? target : '/home';
};

const requireAuth = (req, res, next) => {
  if (req.session && req.session.user) {
    return next();
  }

  const redirectTarget = normalizeRedirectTarget(req);
  if (req.session) {
    req.session.returnTo = redirectTarget;
  }

  if (isApiRequest(req)) {
    return res.status(401).json({
      erro: 'AutenticaÃ§Ã£o necessÃ¡ria. FaÃ§a login para continuar.'
    });
  }

  if (isHtmlRequest(req)) {
    return res.redirect(`/login?redirect=${encodeURIComponent(redirectTarget)}`);
  }

  return res.status(401).json({
    erro: 'AutenticaÃ§Ã£o necessÃ¡ria. FaÃ§a login para continuar.'
  });
};

const requireGuest = (req, res, next) => {
  if (req.session && req.session.user) {
    const redirectTarget = req.session.returnTo || '/home';
    delete req.session.returnTo;
    return res.redirect(redirectTarget);
  }

  next();
};

const requireOwnership = (paramName = 'id') => (req, res, next) => {
  const usuarioLogado = req.session && req.session.user;

  if (!usuarioLogado) {
    return requireAuth(req, res, next);
  }

  const idSolicitado = String(req.params[paramName] || '');
  const idUsuarioLogado = String(usuarioLogado.id || '');

  if (!idSolicitado || idUsuarioLogado === idSolicitado) {
    return next();
  }

  return res.status(403).json({
    erro: 'VocÃª nÃ£o tem permissÃ£o para acessar este recurso.'
  });
};

  requireAuth,
  requireGuest,
  requireOwnership
